 * UPLOAD_FILE handler — injects a file into an <input type="file"> element
 * using a two-phase strategy: set files via native setter + simulate drag-drop.
 */
 
export function handleUploadFile(
  msg: {

    y: number;
    fileName: string;
    mimeType: string;
    base64Data: string;
  },

      const cy = r.top + r.height / 2;
